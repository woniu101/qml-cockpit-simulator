cmake_minimum_required(VERSION 3.21)
project(QmlCockpitSimulator VERSION 0.1 LANGUAGES CXX)
if(MSVC AND CMAKE_CL_SHOWINCLUDES_PREFIX MATCHES "娉|鍖")
    set(CMAKE_CL_SHOWINCLUDES_PREFIX "注意: 包含文件:  ")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Widgets Network Test)
qt_standard_project_setup()
qt_add_executable(VehicleSimulator src/main.cpp src/SimulatorSmoke.cpp src/SimulatorSmoke.h src/SimVehicle.cpp src/SimVehicle.h src/SimServer.cpp src/SimServer.h src/SimulatorWindow.cpp src/SimulatorWindow.h)
target_include_directories(VehicleSimulator PRIVATE src protocol)
target_compile_options(VehicleSimulator PRIVATE /utf-8)
target_link_libraries(VehicleSimulator PRIVATE Qt6::Widgets Qt6::Network Qt6::Test)
set_target_properties(VehicleSimulator PROPERTIES WIN32_EXECUTABLE TRUE)
include(CTest)
if(BUILD_TESTING)
    qt_add_executable(SimulatorTests tests/SimulatorTests.cpp src/SimVehicle.cpp src/SimVehicle.h src/SimServer.cpp src/SimServer.h)
    target_include_directories(SimulatorTests PRIVATE src protocol)
    target_compile_options(SimulatorTests PRIVATE /utf-8)
    target_link_libraries(SimulatorTests PRIVATE Qt6::Core Qt6::Network Qt6::Test)
    add_test(NAME SimulatorTests COMMAND SimulatorTests)
endif()
