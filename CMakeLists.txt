cmake_minimum_required(VERSION 3.21)
project(QmlCockpitSimulator VERSION 0.2.0 LANGUAGES CXX)
if(MSVC AND CMAKE_CL_SHOWINCLUDES_PREFIX MATCHES "娉|鍖")
    set(CMAKE_CL_SHOWINCLUDES_PREFIX "注意: 包含文件:  ")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Widgets Network Test Quick QuickControls2 Quick3D)
qt_standard_project_setup(REQUIRES 6.8)
qt_add_executable(VehicleSimulator src/main.cpp src/SimulatorSmoke.cpp src/SimulatorSmoke.h src/SimVehicle.cpp src/SimVehicle.h src/SimServer.cpp src/SimServer.h src/SimulatorWindow.cpp src/SimulatorWindow.h)
target_sources(VehicleSimulator PRIVATE src/DriverConsole.cpp src/DriverConsole.h)
set_source_files_properties(qml/DriverTheme.qml PROPERTIES QT_QML_SINGLETON_TYPE TRUE)
qt_add_qml_module(VehicleSimulator URI Cockpit.Simulator QML_FILES
    qml/DriverWindow.qml qml/DriverTheme.qml qml/DriverButton.qml qml/DriverPage.qml
    qml/BodyPage.qml qml/ChargePage.qml qml/RoadView.qml qml/KeySettings.qml
    qml/VehiclePlan.qml qml/ChargeStation.qml qml/DiagnosticsPage.qml
    qml/VehicleQuickPanel.qml qml/WindowRocker.qml)
target_include_directories(VehicleSimulator PRIVATE src protocol)
target_compile_options(VehicleSimulator PRIVATE /utf-8)
target_link_libraries(VehicleSimulator PRIVATE Qt6::Widgets Qt6::Network Qt6::Test Qt6::Quick Qt6::QuickControls2 Qt6::Quick3D)
set_target_properties(VehicleSimulator PROPERTIES WIN32_EXECUTABLE TRUE OUTPUT_NAME QmlCockpitSimulator)
if(WIN32)
    set(APP_ICON_PATH "${CMAKE_CURRENT_SOURCE_DIR}/assets/branding/app.ico")
    set(APP_PRODUCT "QML Cockpit Simulator")
    set(APP_FILENAME "QmlCockpitSimulator.exe")
    configure_file(assets/branding/app.rc.in "${CMAKE_CURRENT_BINARY_DIR}/app.rc" @ONLY)
    set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/app.rc" PROPERTIES OBJECT_DEPENDS "${APP_ICON_PATH}")
    target_sources(VehicleSimulator PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/app.rc")
endif()
qt_add_resources(VehicleSimulator branding PREFIX "/branding" BASE "assets/branding" FILES assets/branding/app.ico)
include(CTest)
if(BUILD_TESTING)
    qt_add_executable(SimulatorTests tests/SimulatorTests.cpp src/SimVehicle.cpp src/SimVehicle.h src/SimServer.cpp src/SimServer.h)
    target_include_directories(SimulatorTests PRIVATE src protocol)
    target_compile_options(SimulatorTests PRIVATE /utf-8)
    target_link_libraries(SimulatorTests PRIVATE Qt6::Core Qt6::Network Qt6::Test)
    add_test(NAME SimulatorTests COMMAND SimulatorTests)
endif()
